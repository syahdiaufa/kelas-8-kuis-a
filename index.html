function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var data = []; // Ambil data mulai dari baris kedua (indeks 1) for (var i = 1; i < rows.length; i++) { // Lewati baris jika timestamp dan nama kosong if (!rows[i][0] && !rows[i][1]) continue; data.push({ timestamp: rows[i][0] ? String(rows[i][0]) : "", nama: rows[i][1] || "", kelas: rows[i][2] || "", nilai: rows[i][3] !== undefined ? rows[i][3] : "", jumlahBenar: rows[i][4] !== undefined ? rows[i][4] : "", jumlahSalah: rows[i][5] !== undefined ? rows[i][5] : "", levelDicapai: rows[i][6] !== undefined ? rows[i][6] : "", status: rows[i][7] || "", durasi: rows[i][8] || "", bantuanDipakai: rows[i][9] || "" }); } var callback = e ? e.parameter.callback : null; var json = JSON.stringify({ status: "success", data: data }); if (callback) { return ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data; // Penanganan jika data dikirim via postData JSON atau parameter form if (e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch(err) { data = e.parameter; } } else { data = e.parameter; } sheet.appendRow([ data.timestamp || new Date().toLocaleString("id-ID", { timeZone: "Asia/Makassar" }), data.nama || "-", data.kelas || "Kelas 8", data.nilai !== undefined ? data.nilai : 0, data.jumlahBenar !== undefined ? data.jumlahBenar : 0, data.jumlahSalah !== undefined ? data.jumlahSalah : 0, data.levelDicapai || "-", data.status || "-", data.durasi || "-", data.bantuanDipakai || "-" ]); return ContentService .createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
